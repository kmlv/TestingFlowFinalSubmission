\section{Motivation}
\begin{frame}{
The continuous double auction (CDA) market format}
\begin{itemize}
\item dominated in 20th century financial markets 
\item but design is now problematic.  \pause
\end{itemize}
Problems arise from the conjunction of:
\begin{itemize}
\item strict price/time priority  
\item coarse price grid, and 
\item very low latency telecommunication and computing technology.

 \end{itemize} 
 \end{frame}

\begin{frame}{Creates a ``tilted playing field'': }
incentives to be the first to snap up stale bids/asks may lead to
 \pause
\begin{enumerate}
%\item Unequal opportunities to mitigate price impact
\item unproductive rent-seeking: 10s \$billions on sub-ms speed advantage \pause
\item reduced true liquidity due to toxic takers (``sniping'') \pause
\item increased danger of ``flash crashes,'' as algorithms interact. \pause
  \end{enumerate}
Also tilted by access to complicated ''order shredding'' algorithms.
\end{frame}

\begin{frame}{Reform proposals}
for financial market formats include:
    \begin{itemize}
\item IEX: see e.g., M. Lewis (2014), Aldrich and Friedman (2023); \pause
\item FBA: see e.g., Budish et al (2015), Aldrich and Lopez Vargas (2019); and \pause
\item Flow markets:  Kyle \& Lee (2017), Budish et al. (2023).
\end{itemize}
% The last is my personal favorite, and the subject of today's presentation.
\end{frame}

\begin{frame}{Flow market advocates claim}
that it will bring major benefits:
\begin{enumerate}
\item[0.] Implementable and understandable by human traders.
\item mitigates price impact 
(and volatility) by internalizing order shredding \pause
\item virtually eliminates sniping, and \pause
\item facilitates portfolio trades for interrelated assets.
\end{enumerate}
\end{frame}



\begin{frame}{Our project}

\textbf{compares CDA and Flow} market formats in a simple lab environment  with finely divisible asset units. 
\pause

This environment is 
suitable for testing claims 0 and 1.
\pause

It is \textbf{not} designed to test claims 2 and 3, though it does help set the stage. \pause 

\bigskip

A horse race (CDA vs Flow formats), not a test of theory.\\ \pause 
We compare across formats:
\begin{itemize}
\item traders' behavior \pause
\item market outcomes including 
\begin{itemize}
    \item price dynamics \pause
\item trade volume, and \pause
\item informational and allocative efficiency. 
\end{itemize} 
\end{itemize}


\end{frame}

\section{Market Formats}

\begin{frame}{Continuous Double Auction (CDA) Format}
processes standard limit orders.
\begin{itemize}
    \item These orders are defined by \textbf{three parameters}:$\rightarrow$ ``\underline{Buy/sell} up to \underline{$Q_{max}$} (shares) at a price of \underline{$P$} (dollars per share) or better."
\item Asynchronous order submission within a known time interval. 
\begin{itemize}
\item If a new order does not \textbf{cross} a queued contra-side order, then it is queued in the book. 
\item If it does cross, then it transacts immediately at the best price(s) on the other side of the market.
\end{itemize}
\item Strict \textbf{(price, time) priority}. If there is a tie on price, then older orders are filled first.
    \end{itemize} 
\end{frame}

\begin{frame}{FLOW Market Format}
processes continuous scaled limit orders (CSLOs).
    \begin{itemize}
\item A CSLO has \textbf{five parameters}: $\rightarrow$ ``\underline{Buy/Sell} up to a cumulative total of  \underline{$Q_{max}$} (shares) at maximum rate \underline{$U_{max}$} (shares per second) at prices between \underline{$P_{L}$} and \underline{$P_{H}$} (dollars per share)."
\item Consider a buy order. The (buying rate or) flow demand $U(p(t))$ is the decreasing pw linear function:
\begin{align}
                U{(p(t))}=
                \begin{cases}
                    U_{max} & \text{ if } p{(t)} < P_{L} \\
                    (\frac{P_{H}-p(t)}{P_{H}-P_{L}})U_{max} & \text{ if } P_L \leq p{(t)} \leq P_H \\
                    0  & \text{ if } p{(t)} > P_{H} \\
                \end{cases}
\end{align}
        where $p(t)$ is the clearing price at time $t$.
\item The flow supply (rate) is an analogous pw linear monotone \textbf{increasing} function of $p(t)$
    \end{itemize}
\end{frame}

\begin{frame}{Flow format pricing}
Both the downward sloping demand schedule and upward sloping supply schedule are \textbf{piece-wise linear} functions of price $\rightarrow$ \textbf{unique} price and rate of shares per second. \pause
\begin{itemize}
\item All continuous scaled limit orders ares treated \textbf{symmetrically} and executed \textbf{simultaneously} at the market clearing price. \pause
\item Price changes only when a new order arrives, or an existing order achieves $Q_{max}$ or is cancelled.
\item Traders have \textbf{equal} access to \textbf{opportunities for order shredding}; $P_H - P_L$ and $U_{max}$ are  \textbf{``within-exchange''}
instructions to shred. \pause
\item Consequently, sophisticated algorithms to shred orders would seem unprofitable in the Flow format.
    \end{itemize} %\pause
%Aside: the ``portfolio'' version in Budish et al (2023) is a practical implementation of my monetary exchange process with unified prices. 
\end{frame}

\section{Lab Implementation}

\begin{frame}{Experiment Design: Main Elements}
    \begin{itemize}
        \item Markets of \textbf{8} traders, \textbf{20 paid periods} each (following 2 practice periods). \pause
        \item In each trading period each trader receives a \textbf{buy or sell contract} committing the experimenter to buy or sell up to $x$ shares at some point during the trading period 
        \item e.g., Computer will buy up to 300 shares from you at price 13 ECU at second 115) 
        \item Contracts are private information \pause
        \item simplifications: contracts all sent in first few seconds and are fulfilled in last few seconds; traders' inventories are restricted to $[-Q_{contract}, 0]$ for sellers or $[0, Q_{contract}]$ for buyers.
        \item Subjects can submit or update buy and/or sell orders at any time. \pause
        \item \textbf{Treatments}: \{CDA, FLOW\} (between-group)
        \item Three sets of contracts varied across 5 blocks of 4 periods each. 
    \end{itemize}
\end{frame}

\begin{frame}{Hypotheses}
    \begin{itemize}
\item[H1.] \textit{Traders will submit more but smaller orders in CDA than in Flow format.} \pause
\vspace{0.3cm}
        \item[H2.] \textit{Price volatility will be lower in Flow than in CDA format.}\pause
\vspace{0.3cm}        
        \item[H3.] \textit{Markets will be more liquid [lower PPI] in Flow than in CDA format.} \pause
\vspace{0.3cm}
        \item[H4.] \textit{The CDA and Flow formats will exhibit similar levels of trade volume.} \pause
\vspace{0.3cm}        
        \item[H5.] \textit{The CDA and Flow formats will exhibit similar levels of \\ (a) informational and (b) allocative efficiency.} 
    \end{itemize}
\end{frame}


\section{Results}

\begin{frame}{Summary of Results}

\begin{enumerate}
    \item[R1:] \textbf{Flow} Market format exhibits \textbf{fewer and larger orders}.  \pause 

    
    \item[R2:] \textbf{Flow} format exhibits \textbf{lower price volatility} compared to the CDA.\\  \pause 
   
     \item[R3:] \textbf{PPI} also considerably lower $\implies$ more liquid. \\  \pause 
    \vspace{0.5cm}
    
    \item[R4:] Trade \textbf{Volume is lower in the Flow} Market than in the CDA.  \pause 
    
    \item[R5:] Formats have \textbf{comparable} informational and allocative \textbf{efficiency}. \pause 
  \vspace{0.5cm}
    \item[R6:] (ex post) Buyers' excess profits are far larger than Sellers' in CDA, not so much in Flow. 
    
\end{enumerate}



\end{frame}

\begin{frame}{Price Pressure Index (PPI)}
captures the per-share round trip  cost of buying a standard package ($Q^*=20$ shares) in a time interval ($\Delta t = 5$ sec). \pause
For CDA it is
\begin{equation} \label{eq:ppicda}
    PPI_{CDA} = \frac{1}{Q^*}\int_0^{Q^*} A(Q^{CE}+u,t) - B(Q^{CE}+u,t) du,
\end{equation}
the per-share area between orderbook S and D beyond the CE quantity, averaged over grid times $t=10, 15, ... , 110$ in each period. \pause
For Flow it is
\begin{equation} \label{eq:ppiflow}
    PPI_{Flow} = \frac{1}{Q^*}\int_t^{t+\Delta t} a(r+q,s) - b(r+q,s) ds,
\end{equation}
where $a$ and $b$ are flow S and D (at time $s$), while $r$ is the clearing flow rate (at intersection of $a$ and $b$) and 
$q=\frac{Q^*}{\Delta t}=4$ suffices to acquire the standard package over the time grid interval. Again we report averages over grid times $t=10, 15, ... , 110$ in each period. \pause

Caveat: calculation takes observed order books as given. Actual increases in buying and/or selling may provoke traders to alter their order streams.
\end{frame}



\subsection{Regression Analysis}

\begin{frame}{Regression Specs}
For each outcome metric, we estimate: 
\begin{equation} \label{eq:reg}
    y_{g,t} = \alpha + \sum^{4}_{i = 1} \left(\beta_i \; \text{Block}_i\right) + \gamma \; \text{FLOW}_{g,t} + \theta \; \text{Period}_{g,t} + \epsilon_{g,t}
\end{equation}
where $y_{g,t}$ is the observed outcome  for the given group (i.e., session) and time, and the $\gamma$ coefficient for dummy variable $\text{FLOW}_{g,t}$ is the treatment effect. \pause

We control for: 
\begin{itemize}
\item S-D configuration, via  block fixed effect coefficients $\beta_i$,  
\item time trends, via coefficient $\theta$ for period number $\text{Period}_{g,t}$, and
\item session effects, via clustered s.e.'s.
\end{itemize} \pause
 
% Dependent variables $y$ here are: deviations from CE $ =\mid P^W_t - P_{CE}\mid$, 
% mean absolute interval-to-interval price change $=\mid P^W_{t} - P^W_{t - 1}\mid$, and 
% volatility $=Std\left(\ln P^W_{t} - \ln P^W_{t - 1}\right).$
\end{frame}

\section{Conclusions}

\begin{frame}{Conclusions}
\textbf{First empirical test} of Flow market format, comparing it to CDA in an induced-value setting with finely divisible units. \pause
\begin{itemize}
\item Flow starts worse than CDA but may improve faster (learning).
\item Within exchange \textbf{shredding} seems effective: fewer but larger orders in Flow than in CDA.
\item Price \textbf{volatility} is much lower in Flow. 
\item \textbf{Liquidity} is also much better (PPI lower). \pause
\item Some relative \textbf{undertrading} in Flow.
\item \textbf{Informational  $|P-P_{CE}|$ and allocative efficiencies} are comparable. \pause
\item Buyer-seller \textbf{asymmetry} in Flow seems reduced.\pause
\end{itemize} 
\end{frame}


\begin{frame}{Parting Remarks}

Flow works!
\begin{itemize}
    \item It is comprehensible by human traders, although perhaps they take longer to master it.
    \item It does well even in a simple contract environment that
\begin{itemize}
    \item aligns better with CDA limit orders, and
    \item offers no scope for portfolio trades or protection against predatory HFT traders.
\end{itemize}
\end{itemize}

\section{Appendix}


\begin{frame}{Continuous Double Auction (CDA) Format}
processes standard limit orders.
\begin{itemize}
    \item These orders are defined by \textbf{three parameters}:$\rightarrow$ ``\underline{Buy/sell} up to \underline{$Q_{max}$} (shares) at a price of \underline{$P$} (dollars per share) or better."
\item Asynchronous order submission within a known time interval. 
\begin{itemize}
\item If a new order does not \textbf{cross} a queued contra-side order, then it is queued in the book. 
\item If it does cross, then it transacts immediately at the best price(s) on the other side of the market.
\end{itemize}
\item Strict \textbf{(price, time) priority}. If there is a tie on price, then older orders are filled first.
    \end{itemize} 
\end{frame}



\begin{frame}{CDA Timing Issues}
1. Early bird special  
    \begin{itemize}
        \item Since market supply and demand are discontinuous step functions of price, there is \textbf{no guarantee of a single point of intersection}: there can be excess of demand (supply), and often there is. 
        \item Hence, \textbf{an allocation rule} to determine $\alpha(t_0, t)$ is needed (e.g., time priority, or pro-rata)
\item In most existing financial markets, orders are \textbf{prioritized} based on \textbf{the time they arrive}, so high speed is rewarded
\item If an agent can \textbf{react to a relevant signal} (even if public) \textbf{faster} (even if only a tiny bit) than other traders, then she can \textbf{pick off stale orders} from other traders and reverse position after the information has spread in the market.
\item  Thus: tilted playing field, rent-seeking, toxic takers $\implies$ illiquidity, increased flash crash risk.
    \end{itemize}
    2. To avoid adverse price pressure, traders will want to \textbf{shred} large orders.
    \end{frame}
